def run_elevated(params):

    sei.cbSize = ctypes.sizeof(sei)
    sei.fMask = 0x00000040
    sei.lpVerb = "runas"

    sei.lpDirectory = HERE
    sei.nShow = 1

        return "ShellExecuteExW failed err=%d" % ctypes.get_last_error()

    kernel32.GetExitCodeProcess(sei.hProcess, ctypes.byref(code))
    kernel32.CloseHandle(sei.hProcess)
    return "elevated child finished, exit=%d" % code.value

def find_pid():
    snap = kernel32.CreateToolhelp32Snapshot(0x00000002, 0)
    if snap == ctypes.c_void_p(-1).value:
        return None
    pe = PROCESSENTRY32W()
    pe.dwSize = ctypes.sizeof(pe)
    ok = kernel32.Process32FirstW(snap, ctypes.byref(pe))
    hit = None
    while ok:
        if pe.szExeFile.lower().startswith(PROC_NAME):
            hit = pe.th32ProcessID
            break
        ok = kernel32.Process32NextW(snap, ctypes.byref(pe))
    kernel32.CloseHandle(snap)
    return hit

def scan_ranges(ram_top, lo=None, hi=None):
    top = min(ram_top, 0x4000000000) if ram_top else 0x100000000
    ranges = []
    cur = SCAN_LO
    for rlo, rhi in sorted(HOLES):
        if rhi <= cur:
            continue
        if rlo >= top:
            break
        if rlo > cur:
            ranges.append((cur, min(rlo, top)))
        cur = max(cur, rhi)
        if cur >= top:
            break
    if cur < top:
        ranges.append((cur, top))
    if lo is not None or hi is not None:
        lo = SCAN_LO if lo is None else lo
        hi = top if hi is None else min(hi, top)
        clipped = []
        for rlo, rhi in ranges:
            a, b = max(rlo, lo), min(rhi, hi)
            if b > a:
                clipped.append((a, b))
        ranges = clipped
    return ranges


def name_hits(g, ram_top, chunk=SCAN_CHUNK, cap=HIT_CAP, lo=None, hi=None):
    """Physical addresses of the NAME string; faults skip one chunk.
    A 15-byte carry covers hits straddling chunk boundaries.  Every hit and a
    periodic checkpoint go straight to PROG, so a box hang keeps the evidence."""
    hits = []
    skipped = 0
    chunks = 0
    scanned = 0
    nxt_ck = CKPT_BYTES
    for rlo, rhi in scan_ranges(ram_top, lo, hi):
        pa = rlo
        carry = b""
        while pa < rhi and len(hits) < cap:
            n = min(chunk, rhi - pa)
            try:
                blk = g.read_phys(pa, n)
            except OSError:
                skipped += 1
                chunks += 1
                prog("FAULT pa=%#x n=%#x" % (pa, n))
                pa += chunk
                carry = b""
            else:
                chunks += 1
                buf2 = carry + blk
                base = pa - len(carry)
                off = 0
                while len(hits) < cap:
                    i = buf2.find(NAME, off)
                    if i < 0:
                        break
                    hits.append(base + i)
                    prog("HIT pa=%#x (%d)" % (base + i, len(hits)))
                    off = i + 1
                carry = blk[-15:]
                pa += n
            scanned += n
            if scanned >= nxt_ck:
                prog("CKPT pa=%#x scanned=%d MiB chunks=%d hits=%d fault=%d %.1fs"
                     % (pa, scanned >> 20, chunks, len(hits), skipped,
                        time.perf_counter() - _T0))
                nxt_ck += CKPT_BYTES
    prog("SCANEND scanned=%d MiB chunks=%d hits=%d fault=%d %.1fs"
         % (scanned >> 20, chunks, len(hits), skipped,
            time.perf_counter() - _T0))
    return hits, skipped, chunks


def valid_cr3(g, dtb):
    return (dtb and not (dtb & 0xFFF) and dtb < g.ram_top and g.is_ram(dtb))


def candidates_for(g, pid, hits):
    """Dedupe cr3 candidates: name page -> shift -> pid dword -> DTB."""
    out = []
    seen = set()
    for page in dict.fromkeys(h & ~0xFFF for h in hits):
        try:
            p = g.read_phys(page, 0x2000)      # page + successor (header+pid)
        except OSError:
            continue
        off = next(h for h in hits if (h & ~0xFFF) == page) - page
        for s in SHIFTS:
            b = off + s
            if b + 0x488 > len(p):
                continue
            if not any(struct.unpack_from("<I", p, b + o)[0] == pid
                       for o in PID_OFFS):
                continue
            dtb = struct.unpack_from("<Q", p, b + 0x28)[0]
            if not valid_cr3(g, dtb):
                log("  cand page=%#x shift=%#x pid-ok dtb=%#x (reject)"
                    % (page, s, dtb))
                continue
            key = (page, s)
            if key in seen:
                continue
            seen.add(key)
            out.append((page + s, dtb))
    if not out:
        log("  %d hit pages, none passed the pid gate" % len(hits))
    return out

def validate_pe(buf):
    """Strict PE32+ header check on a 4 KiB page buffer."""
    if len(buf) < 0x1000 or buf[:2] != b"MZ":
        return None
    elf = struct.unpack_from("<I", buf, 0x3C)[0]
    if not (0x40 <= elf <= 0xF00) or buf[elf:elf + 4] != b"PE\x00\x00":
        return None
    machine, nsec = struct.unpack_from("<HH", buf, elf + 4)
    sizeopt = struct.unpack_from("<H", buf, elf + 20)[0]
    if machine != 0x8664 or not (1 <= nsec <= 40) or sizeopt < 0xF0:
        return None
    if struct.unpack_from("<H", buf, elf + 24)[0] != 0x20B:
        return None
    salign, sizeimg, sizehdr = struct.unpack_from("<III", buf, elf + 32)[0], \
        struct.unpack_from("<I", buf, elf + 56)[0], \
        struct.unpack_from("<I", buf, elf + 60)[0]
    if salign != 0x1000 or not (MIN_IMAGE <= sizeimg <= MAX_IMAGE):
        return None
    if not (0x200 <= sizehdr <= 0x1000):
        return None
    if elf + 24 + sizeopt + 40 * nsec > 0x1000:
        return None
    secs = []
    so = elf + 24 + sizeopt
    for i in range(nsec):
        o = so + 40 * i
        name = buf[o:o + 8].rstrip(b"\x00").decode("latin-1")
        vsize, va, rsize = struct.unpack_from("<III", buf, o + 8)
        vsize = vsize or rsize
        if va >= sizeimg:
            continue
        vsize = min(vsize, sizeimg - va)
        secs.append((name, va, vsize))
    return {"size": sizeimg, "hdr": sizehdr, "secs": secs}

def u64(buf, off):
    return struct.unpack_from("<Q", buf, off)[0]


class Walk(object):
    """CR3 -> VA page resolution; PD pages cached, every PA is_ram-checked."""

    def __init__(self, g, cr3):
        self.g = g
        self.cr3 = cr3
        self.pdpt_pa = 0
        self.cache = {}
        pm = g.read_phys(cr3, 0x1000)          # PML4
        e0 = u64(pm, 0)
        if e0 & 1:
            p = e0 & MASK
            if g.is_ram(p):
                self.pdpt_pa = p

    def pt_for(self, va):
        if not self.pdpt_pa:
            return None
        ii = (va >> 30) & 0x1FF
        jj = (va >> 21) & 0x1FF
        key = (ii, jj)
        if key in self.cache:
            return self.cache[key]
        res = None
        try:
            e = self.g.u64(self.pdpt_pa + ii * 8)
            if (e & 1) and not (e & 0x80):
                pd_pa = e & MASK
                if self.g.is_ram(pd_pa):
                    pd = self.g.read_phys(pd_pa, 0x1000)
                    d = u64(pd, jj * 8)
                    if (d & 1) and not (d & 0x80):
                        pt_pa = d & MASK
                        if self.g.is_ram(pt_pa):
                            res = self.g.read_phys(pt_pa, 0x1000)
        except OSError:
            res = None
        self.cache[key] = res
        return res

    def page_pa(self, va):
        pt = self.pt_for(va)
        if pt is None:
            return None
        pte = u64(pt, ((va >> 12) & 0x1FF) * 8)
        if not (pte & 1):
            return None
        pa = pte & MASK
        return pa if self.g.is_ram(pa) else None

def alloc_locked(size):
    p = kernel32.VirtualAlloc(None, size, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE)
    if not p:
        raise OSError("VirtualAlloc failed err=%d" % ctypes.get_last_error())
    kernel32.VirtualLock(p, size)
    return p


class Gio(object):
    """One device handle + a scratch landing buffer for phys reads."""

    def __init__(self, path=DEVICE, chunk=PAGE, map_cache_pages=64):
        self.path = path
        self.chunk = chunk
        self.h = kernel32.CreateFileW(path, GENERIC_READ | GENERIC_WRITE, 0,
                                      None, OPEN_EXISTING, 0, None)
        if not self.h or self.h == INVALID_HANDLE:
            raise OSError(ctypes.get_last_error(),
                          "CreateFileW %s failed (need elevation)" % path)
        self.dst = kernel32.VirtualAlloc(None, chunk, MEM_COMMIT | MEM_RESERVE,
                                         PAGE_READWRITE)
        if not self.dst:
            raise OSError("VirtualAlloc scratch failed")
        kernel32.VirtualLock(self.dst, chunk)
        self.ram_top = installed_ram_bytes()
        self.map_cache = {}          # page_pa -> (va, size)
        self.map_cache_max = map_cache_pages
        self.stats = {}              # ioctl -> [count, seconds]

    # -- low level ---------------------------------------------------------
    def _ioctl(self, code, in_buf, out_len, out_hint=64):
        ib = ctypes.create_string_buffer(bytes(in_buf), max(len(in_buf), 1))
        ob = ctypes.create_string_buffer(max(out_len, out_hint, 1))
        got = wintypes.DWORD(0)
        t0 = time.perf_counter()
        ok = kernel32.DeviceIoControl(self.h, code, ib, len(in_buf), ob,
                                      out_len, ctypes.byref(got), None)
        dt = time.perf_counter() - t0
        st = self.stats.setdefault(code, [0, 0.0])
        st[0] += 1
        st[1] += dt
        if not ok:
            return False, ctypes.get_last_error(), b"", got.value
        return True, 0, ob.raw[:got.value], got.value

    # -- read (verified primitive) ------------------------------------------
    def read_phys(self, pa, size):
        out = bytearray()
        left = size
        cur = pa
        while left:
            n = min(left, self.chunk)
            ib = struct.pack("<QIIQ", cur & 0xFFFFFFFFFFFFFFFF, n, 0, self.dst)
            ok, err, _, _ = self._ioctl(IOCTL_READ, ib, 0)
            if not ok:
                raise OSError(err, "phys read %#x+%#x failed" % (cur, n))
            out += ctypes.string_at(self.dst, n)
            cur += n
            left -= n
        return bytes(out)

    def u16(self, pa):
        return struct.unpack("<H", self.read_phys(pa, 2))[0]

    def u32(self, pa):
        return struct.unpack("<I", self.read_phys(pa, 4))[0]

    def u64(self, pa):
        return struct.unpack("<Q", self.read_phys(pa, 8))[0]

    # -- map / unmap (write path candidate) ----------------------------------
    def map_phys(self, pa, size):
        ib = struct.pack("<QQ", pa, size)
        ok, err, ob, got = self._ioctl(IOCTL_MAP_EX, ib, 0x10)
        if not ok:
            raise OSError(err, "map_phys %#x+%#x failed" % (pa, size))
        if got < 8:
            raise OSError("map_phys returned short info %d" % got)
        va = struct.unpack_from("<QQ", ob)[0]
        if not va:
            raise OSError("map_phys %#x+%#x -> va=0" % (pa, size))
        return va

    def unmap(self, va, size):
        ib = struct.pack("<QQ", va, size)
        ok, err, _, _ = self._ioctl(IOCTL_UNMAP, ib, 0)
        if not ok:
