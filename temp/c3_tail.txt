 200|     log("no candidate  tried=%d shapes=%d %.1fs"
 201|         % (tried, shapes, time.perf_counter() - t0))
 202|     return None, None, "none"

 205| def hop(g, b, va, seen):
 206|     """VA -> (pid, base_pa, fl, bl, name) for a list node, or None.
 207| 
 208|     p44d: the node VA is base+0x258, so the object starts at pa-0x258 -- NOT the
 209|     -0x1d8 lane the decoy's fake LINK_ENTRY sits on.
 210|     """
 211|     base = va - b - OBJ_LINK
 212|     o = base & 0xFFF
 213|     if base in seen or not g.is_ram(base) or o > 0xCA0:
 214|         return None
 215|     seen.add(base)
 216|     try:
 217|         p = g.read_phys(base - o, 0x1000)
 218|     except OSError:
 219|         return None
 220|     if p[o] != 3:
 221|         return None
 222|     pid = struct.unpack_from("<I", p, o + OBJ_PID)[0]
 223|     fl, bl = struct.unpack_from("<QQ", p, o + OBJ_LINK)
 224|     return pid, base, fl, bl, p[o + OBJ_NAME:o + OBJ_NAME + 15]

 227| def walk(g, b, start_va, seen, tag):
 228|     """Follow one direction; collect (pid, base, name) until HOP_CAP or a repeat."""
 229|     va = start_va
 230|     out = []
 231|     for i in range(HOP_CAP):
 232|         h = hop(g, b, va, seen)
 233|         if not h:
 234|             out.append(("dead", va))
 235|             break
 236|         pid, base, fl, bl, nm = h
 237|         out.append((pid, base, nm))
 238|         prog("%s hop%d pid=%#x base=%#x name=%r" % (tag, i, pid, base, nm))
 239|         nxt = fl if tag == "FL" else bl
 240|         if nxt == start_va:
 241|             out.append(("ring", nxt))
 242|             break
 243|         va = nxt
 244|     return out

 247| def main():
 248|     sys.argv = ["basehunt"] + sys.argv[1:]
 249|     dry = "--dry" in sys.argv
 250|     start = None
 251|     budget = 0.0
 252|     for a in sys.argv[1:]:
 253|         if a.startswith("--from="):
 254|             start = int(a.split("=", 1)[1], 0)
 255|         elif a.startswith("--max-s="):
 256|             budget = float(a.split("=", 1)[1])
 257|     if not dl.elevation():
 258|         return wr(3, "NOT_ELEVATED")
 259|     log("basehunt  %s%s" % (time.strftime("%Y-%m-%d %H:%M:%S"), "  DRY" if dry else ""))
 260|     try:
 261|         g = gio.Gio(chunk=dl.SCAN_CHUNK)
 262|     except OSError as e:
 263|         log("FATAL: %s" % e)
 264|         return wr(2, "FATAL no transport")
 265|     try:
 266|         ranges, note = dl.usable_ranges(g.ram_top, None, None)
 267|         log("RAMTOP %#x %s ranges=%d" % (g.ram_top, note, len(ranges)))
 268|         b = hunt(g, NODE_FL, g.ram_top, start, budget)
 269|         if b is None:
 270|             return wr(1, "NO_BASE")
 271|         if dry:
 272|             return wr(0, "DRY_ACCEPT b=%#x" % b)
 273|         seen = set()
 274|         fl = walk(g, b, NODE_FL, seen, "FL")
 275|         bl = walk(g, b, NODE_BL, seen, "BL")
 276|         log("FL %d hops, BL %d hops" % (len(fl), len(bl)))
 277|         targets = [x for x in fl + bl if isinstance(x[0], int) and x[0] == 0x1570]
 278|         log("pid 0x1570 hops: %d" % len(targets))
 279|         import ctypes
 280|         buf = ctypes.create_string_buffer(0x2000000)
 281|         st = dl.Stats()
 282|         for pid, base, nm in targets:
 283|             try:
 284|                 p = g.read_phys(base, 0x1000)
 285|             except OSError:
 286|                 log("  base %#x unreadable" % base)
 287|                 continue
 288|             dtb = struct.unpack_from("<Q", p, 0x28)[0] & dl.MASK
 289|             log("  target base=%#x dtb=%#x name=%r" % (base, dtb, nm))
 290|             if not g.is_ram(dtb):
 291|                 log("   dtb outside RAM")
 292|                 continue
 293|             why = dl.probe_dtb(g, dtb)
 294|             log("   probe: %s" % why)
 295|             if "pdpt ram" not in why:
 296|                 log("   dtb does not page")
 297|                 continue
 298|             mz = dl.find_mz(g, dtb)
 299|             if not mz:
 300|                 log("   no MZ")
 301|                 continue
 302|             mva, meta = mz
 303|             log("   MZ va=%#x size=%#x secs=%d" % (mva, meta["size"], len(meta["secs"])))
 304|             w = dl.Walk(g, dtb)
 305|             for name, sva, vsize in meta["secs"]:
 306|                 log("   sec %-8s va=%#x size=%#x" % (name, sva, vsize))
 307|                 dl.dump_range(g, w, buf, mva, sva, min(vsize, 0x2000000), st)
 308|             log("   read=%#x zero=%#x fail=%#x" % (st.read, st.zero, st.fail))
 309|             path = os.path.join(ROOT, "_ritual", "p44f_image.bin")
 310|             open(path, "wb").write(bytes(buf))
 311|             import hashlib
 312|             h = hashlib.sha256(bytes(buf)).hexdigest()
 313|             log("   dump %s sha256=%s" % (path, h))
 314|             return wr(0, "DUMP_OK pid=0x1570 base=%#x dtb=%#x va=%#x sha256=%s"
 315|                       % (base, dtb, mva, h))
 316|         return wr(1, "NO_LIVE_OBJECT fl=%d bl=%d" % (len(fl), len(bl)))
 317|     finally:
 318|         g.close()
 319|

 321| if __name__ == "__main__":
 322|     sys.exit(main())
