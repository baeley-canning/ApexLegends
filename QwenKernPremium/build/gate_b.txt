96: function Get-StaticImports([string] $dumpbin, [string] $artifact) {
97:     $output  = @(& $dumpbin /nologo /dependents $artifact 2>&1)
98:     $imports = New-Object System.Collections.Generic.List[string]
99:     $inBlock = $false
100:     foreach ($line in $output) {
101:         $trimmed = "$line".Trim()
102:         if ($trimmed -match '^Image has the following dependencies:') { $inBlock = $true;  continue }
103:         if ($trimmed -match 'following delay load dependencies:')      { $inBlock = $false; continue }
104:         if (-not $inBlock) { continue }
105:         if ($trimmed -eq '') {
106:             # dumpbin prints a blank line immediately AFTER the header, so a
107:             # blank line only closes the block once names have been collected.
108:             if ($imports.Count -gt 0) { $inBlock = $false }
109:             continue
110:         }
111:         $imports.Add($trimmed)
112:     }
113:     return $imports
