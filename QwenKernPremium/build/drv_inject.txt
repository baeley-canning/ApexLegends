148: // ----------------------------------------------------------------------------
149: static NTSTATUS ApexdrvInject(PAPEXDRV_INJECT_REQUEST request) {
150:     HANDLE   processHandle = NULL;
151:     HANDLE   threadHandle  = NULL;
152:     PVOID    remoteBuffer  = NULL;
153:     SIZE_T   regionSize    = 0;
154:     SIZE_T   pathBytes     = 0;
155:     NTSTATUS status;
156: 
157:     if (request->ProcessId == 0)          return STATUS_INVALID_PARAMETER;
158:     if (request->LoadLibraryAddress == 0)  return STATUS_INVALID_PARAMETER;
159: 
160:     // Ensure the wide path is NUL terminated inside the supplied buffer.
161:     request->DllPath[APEXDRV_MAX_PATH_CHARS - 1] = L'\0';
162:     pathBytes = (wcslen(request->DllPath) + 1) * sizeof(WCHAR);
163:     if (pathBytes <= sizeof(WCHAR)) return STATUS_INVALID_PARAMETER;
164: 
165:     status = ApexdrvOpenProcess(request->ProcessId, &processHandle);
166:     if (!NT_SUCCESS(status)) return status;
167: 
168:     regionSize = pathBytes;
169:     status = ZwAllocateVirtualMemory(processHandle, &remoteBuffer, 0, &regionSize,
170:                                      MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
171:     if (!NT_SUCCESS(status)) {
172:         ZwClose(processHandle);
173:         return status;
174:     }
175: 
176:     status = ZwWriteVirtualMemory(processHandle, remoteBuffer,
177:                                   request->DllPath, pathBytes, NULL);
178:     if (!NT_SUCCESS(status)) {
179:         ZwFreeVirtualMemory(processHandle, &remoteBuffer, &regionSize, MEM_RELEASE);
180:         ZwClose(processHandle);
181:         return status;
182:     }
183: 
184:     // User-mode thread running kernel32!LoadLibraryW(remotePath).
185:     status = ZwCreateThreadEx(&threadHandle, THREAD_ALL_ACCESS, NULL, processHandle,
186:                               (PVOID)(ULONG_PTR)request->LoadLibraryAddress,
187:                               remoteBuffer, 0, 0, 0, 0, NULL);
188:     if (!NT_SUCCESS(status)) {
189:         ZwFreeVirtualMemory(processHandle, &remoteBuffer, &regionSize, MEM_RELEASE);
190:         ZwClose(processHandle);
191:         return status;
192:     }
193: 
194:     {
195:         PETHREAD threadObject = NULL;
196:         if (NT_SUCCESS(ObReferenceObjectByHandle(threadHandle,
197:                                                  THREAD_QUERY_INFORMATION,
198:                                                  *PsThreadType, KernelMode,
199:                                                  (PVOID*)&threadObject, NULL))) {
200:             request->ThreadId = (ULONG)(ULONG_PTR)PsGetThreadId(threadObject);
201:             ObDereferenceObject(threadObject);
202:         }
203:     }
204: 
205:     // Wait for the loader so the path buffer can safely be released.
206:     {
207:         LARGE_INTEGER timeout;
208:         timeout.QuadPart = -((LONGLONG)APEXDRV_MAX_THREAD_WAIT_MS * 10000);
209:         KeWaitForSingleObject(threadHandle, Executive, KernelMode, FALSE, &timeout);
210:     }
211: 
212:     request->RemoteBase   = (ULONG64)(ULONG_PTR)remoteBuffer;
213:     request->ResultStatus = STATUS_SUCCESS;
214: 
215:     ZwClose(threadHandle);
216:     ZwFreeVirtualMemory(processHandle, &remoteBuffer, &regionSize, MEM_RELEASE);
217:     ZwClose(processHandle);
218: 
219:     return STATUS_SUCCESS;
220: }
