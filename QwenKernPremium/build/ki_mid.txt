102:     SC_HANDLE service = OpenServiceW(scm, APEXDRV_SERVICE_NAME,
103:                                      SERVICE_STOP | SERVICE_QUERY_STATUS | DELETE);
104:     if (!service) return;
105: 
106:     SERVICE_STATUS status{};
107:     ControlService(service, SERVICE_CONTROL_STOP, &status);
108:

110:     for (int i = 0; i < 20; ++i) {
111:         if (!QueryServiceStatus(service, &status) ||
112:             status.dwCurrentState == SERVICE_STOPPED) {
113:             break;
114:         }
115:         Sleep(50);
116:     }
