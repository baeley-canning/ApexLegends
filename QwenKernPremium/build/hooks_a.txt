113: // ---------------------------------------------------------------------------
114: // Entry points resolved on demand.
115: //
116: // d3d9.dll and d3dcompiler_47.dll are NOT loaded by r5apex.exe.  Importing
117: // them statically would both inject two foreign modules into the game and put
118: // them in the IAT - exactly what the anti-cheat's module whitelist inspects.
119: // ---------------------------------------------------------------------------
120: typedef HRESULT (WINAPI *D3DCompileFn)(LPCVOID, SIZE_T, LPCSTR,
121:                                        const D3D_SHADER_MACRO*, ID3DInclude*,
122:                                        LPCSTR, LPCSTR, UINT, UINT,
123:                                        ID3DBlob**, ID3DBlob**);
124: 
125: D3DCompileFn ResolveD3DCompile() {
126:     static D3DCompileFn cached = nullptr;
127:     if (cached) return cached;
128: 
129:     HMODULE compiler = LoadLibraryA("d3dcompiler_47.dll");
130:     if (!compiler) return nullptr;
131: 
132:     cached = reinterpret_cast<D3DCompileFn>(GetProcAddress(compiler, "D3DCompile"));
133:     return cached;
134: }
135: 
136: typedef IDirect3D9* (WINAPI *Direct3DCreate9Fn)(UINT);
137: 
138: Direct3DCreate9Fn ResolveDirect3DCreate9() {
139:     static Direct3DCreate9Fn cached = nullptr;
140:     if (cached) return cached;
141: 
142:     HMODULE d3d9Module = LoadLibraryA("d3d9.dll");
143:     if (!d3d9Module) return nullptr;
144: 
145:     cached = reinterpret_cast<Direct3DCreate9Fn>(
146:         GetProcAddress(d3d9Module, "Direct3DCreate9"));
147:     return cached;
148: }

150: void ReleaseD3D11Overlay() {
151:     if (g_overlayBuffer) { g_overlayBuffer->Release(); g_overlayBuffer = nullptr; }
152:     if (g_overlayLayout) { g_overlayLayout->Release(); g_overlayLayout = nullptr; }
153:     if (g_overlayPS)     { g_overlayPS->Release();     g_overlayPS     = nullptr; }
154:     if (g_overlayVS)     { g_overlayVS->Release();     g_overlayVS     = nullptr; }
155:     if (g_overlayBlend)  { g_overlayBlend->Release();  g_overlayBlend  = nullptr; }
156:     if (g_overlayDepth)  { g_overlayDepth->Release();  g_overlayDepth  = nullptr; }
157:     if (g_overlayRaster) { g_overlayRaster->Release(); g_overlayRaster = nullptr; }
158:     if (g_d3d11Context)  { g_d3d11Context->Release();  g_d3d11Context  = nullptr; }
159:     if (g_d3d11Device)   { g_d3d11Device->Release();   g_d3d11Device   = nullptr; }
160:     g_d3d11Ready      = false;
161:     g_overlayCapacity = 0;
162: }

166: // ============================================================================
167: // D3D11 pipeline (re)creation â€” called lazily from the Present thunk, i.e. on
168: // the first frame the game actually presents.
169: // ============================================================================
170: static bool EnsureD3D11Overlay(IDXGISwapChain* swapChain) {
171:     if (g_d3d11Ready) return true;
172:     if (!swapChain) return false;
173: 
174:     if (FAILED(swapChain->GetDevice(__uuidof(ID3D11Device),
175:                                     reinterpret_cast<void**>(&g_d3d11Device)))) {
176:         return false;
177:     }
178:     g_d3d11Device->GetImmediateContext(&g_d3d11Context);
179: 
180:     const D3DCompileFn compile = ResolveD3DCompile();
181:     if (!compile) {
182:         ReleaseD3D11Overlay();
183:         return false;
184:     }
185: 
186:     ID3DBlob* vsBlob = nullptr;
187:     ID3DBlob* psBlob = nullptr;
188:     ID3DBlob* errors = nullptr;
189: 
190:     const bool vsOk = SUCCEEDED(compile(kOverlayShaderSource,
191:                                             sizeof(kOverlayShaderSource) - 1,
192:                                             "apex_overlay", nullptr, nullptr,
193:                                             "VSMain", "vs_4_0",
194:                                             D3DCOMPILE_OPTIMIZATION_LEVEL3, 0,
195:                                             &vsBlob, &errors));
196:     const bool psOk = SUCCEEDED(compile(kOverlayShaderSource,
197:                                             sizeof(kOverlayShaderSource) - 1,
198:                                             "apex_overlay", nullptr, nullptr,
199:                                             "PSMain", "ps_4_0",
200:                                             D3DCOMPILE_OPTIMIZATION_LEVEL3, 0,
201:                                             &psBlob, &errors));
202:     if (errors) { errors->Release(); errors = nullptr; }
203: 
204:     if (!vsOk || !psOk) {
205:         if (vsBlob) vsBlob->Release();
206:         if (psBlob) psBlob->Release();
207:         ReleaseD3D11Overlay();
208:         return false;
209:     }
210: 
211:     bool ok = SUCCEEDED(g_d3d11Device->CreateVertexShader(vsBlob->GetBufferPointer(),
212:                                                           vsBlob->GetBufferSize(),
213:                                                           nullptr, &g_overlayVS)) &&
214:               SUCCEEDED(g_d3d11Device->CreatePixelShader(psBlob->GetBufferPointer(),
215:                                                          psBlob->GetBufferSize(),
216:                                                          nullptr, &g_overlayPS));
217: 
218:     const D3D11_INPUT_ELEMENT_DESC layout[] = {
219:         { "POSITION", 0, DXGI_FORMAT_R32G32_FLOAT,   0, 0,
220:           D3D11_INPUT_PER_VERTEX_DATA, 0 },
221:         { "COLOR",    0, DXGI_FORMAT_R8G8B8A8_UNORM, 0, 8,
222:           D3D11_INPUT_PER_VERTEX_DATA, 0 },
223:         { "TEXCOORD", 0, DXGI_FORMAT_R32G32_FLOAT,   0, 12,
224:           D3D11_INPUT_PER_VERTEX_DATA, 0 },
225:     };
226: 
227:     ok = ok && SUCCEEDED(g_d3d11Device->CreateInputLayout(
228:         layout, ARRAYSIZE(layout), vsBlob->GetBufferPointer(),
229:         vsBlob->GetBufferSize(), &g_overlayLayout));
230: 
231:     vsBlob->Release();
232:     psBlob->Release();
233: 
234:     g_overlayCapacity = 64 * 8;   // 64 boxes x 8 vertices (4 lines per box)
235: 
236:     D3D11_BUFFER_DESC bufferDesc{};
237:     bufferDesc.ByteWidth      = g_overlayCapacity * sizeof(D3D11Vertex);
238:     bufferDesc.Usage          = D3D11_USAGE_DYNAMIC;
239:     bufferDesc.BindFlags      = D3D11_BIND_VERTEX_BUFFER;
240:     bufferDesc.CPUAccessFlags = D3D11_CPU_ACCESS_WRITE;
241:     ok = ok && SUCCEEDED(g_d3d11Device->CreateBuffer(&bufferDesc, nullptr,
242:                                                       &g_overlayBuffer));
243: 
244:     D3D11_BLEND_DESC blendDesc{};
245:     blendDesc.RenderTarget[0].BlendEnable           = TRUE;
246:     blendDesc.RenderTarget[0].SrcBlend              = D3D11_BLEND_SRC_ALPHA;
247:     blendDesc.RenderTarget[0].DestBlend             = D3D11_BLEND_INV_SRC_ALPHA;
248:     blendDesc.RenderTarget[0].BlendOp               = D3D11_BLEND_OP_ADD;
249:     blendDesc.RenderTarget[0].SrcBlendAlpha         = D3D11_BLEND_ONE;
250:     blendDesc.RenderTarget[0].DestBlendAlpha        = D3D11_BLEND_ZERO;
251:     blendDesc.RenderTarget[0].BlendOpAlpha          = D3D11_BLEND_OP_ADD;
252:     blendDesc.RenderTarget[0].RenderTargetWriteMask = D3D11_COLOR_WRITE_ENABLE_ALL;
253:     ok = ok && SUCCEEDED(g_d3d11Device->CreateBlendState(&blendDesc, &g_overlayBlend));
254: 
255:     D3D11_DEPTH_STENCIL_DESC depthDesc{};
256:     depthDesc.DepthEnable = FALSE;
257:     ok = ok && SUCCEEDED(g_d3d11Device->CreateDepthStencilState(&depthDesc,
258:                                                                 &g_overlayDepth));
259: 
260:     D3D11_RASTERIZER_DESC rasterDesc{};
261:     rasterDesc.FillMode        = D3D11_FILL_SOLID;
262:     rasterDesc.CullMode        = D3D11_CULL_NONE;
263:     rasterDesc.DepthClipEnable = TRUE;
264:     ok = ok && SUCCEEDED(g_d3d11Device->CreateRasterizerState(&rasterDesc,
265:                                                               &g_overlayRaster));
266: 
267:     if (!ok) {
268:         ReleaseD3D11Overlay();
269:         return false;
270:     }
271: 
272:     g_d3d11Ready = true;
273:     return true;
274: }

279: bool GameHooks::InstallHooks(MemoryReader* reader) {
280:     if (bHooksInstalled || !reader)
281:         return false;
282: 
283:     pMemoryReader = reader;
284:     FindGameWindow();
285: 
286:     // Apex Legends is a DX11 title: hook IDXGISwapChain::Present first.  The
287:     // D3D9 path is only used when d3d9.dll is already loaded in the process
288:     // (loading it ourselves would be an obvious module artifact).
289:     bDXGIHooked = HookPresentDXGI();
290:     if (!bDXGIHooked && GetModuleHandleA("d3d9.dll") != nullptr) {
291:         bPresentHooked = HookPresent(nullptr);
292:     }
293: 
294:     if (!bDXGIHooked && !bPresentHooked) {
295:         printf("[-] No render backend could be hooked\n");
296:     } else {
297:         printf("[+] Render hook installed (%s Present)\n",
298:                bDXGIHooked ? "DXGI" : "D3D9");
299:     }
300: 
301:     // Subclass the game window (overlay toggle keys / input filtering).
302:     if (hGameWindow) {
303:         originalWndProc = reinterpret_cast<WNDPROC>(SetWindowLongPtrW(
304:             hGameWindow, GWLP_WNDPROC,
305:             reinterpret_cast<LONG_PTR>(&GameHooks::WndProcHook)));
306:     }
307: 
308:     bHooksInstalled = true;
309:     return bDXGIHooked || bPresentHooked;
310: }
311: 
312: void GameHooks::UninstallHooks() {
313:     if (!bHooksInstalled)
314:         return;
315: 
316:     if (originalWndProc && hGameWindow) {
317:         SetWindowLongPtrW(hGameWindow, GWLP_WNDPROC,
318:                           reinterpret_cast<LONG_PTR>(originalWndProc));
319:         originalWndProc = nullptr;
320:     }
321: 
322:     DWORD oldProtect = 0;
323:     if (g_dxgiPresentSlot && originalDXGIPresent) {
324:         if (VirtualProtect(g_dxgiPresentSlot, sizeof(void*),
325:                            PAGE_EXECUTE_READWRITE, &oldProtect)) {
326:             *g_dxgiPresentSlot = reinterpret_cast<void*>(originalDXGIPresent);
327:             VirtualProtect(g_dxgiPresentSlot, sizeof(void*), oldProtect, &oldProtect);
328:         }
329:         originalDXGIPresent = nullptr;
330:     }
331:     if (g_d3d9PresentSlot && originalPresent) {
332:         if (VirtualProtect(g_d3d9PresentSlot, sizeof(void*),
333:                            PAGE_EXECUTE_READWRITE, &oldProtect)) {
334:             *g_d3d9PresentSlot = reinterpret_cast<void*>(originalPresent);
335:             VirtualProtect(g_d3d9PresentSlot, sizeof(void*), oldProtect, &oldProtect);
336:         }
337:         originalPresent = nullptr;
338:     }
339: 
340:     ReleaseD3D11Overlay();
