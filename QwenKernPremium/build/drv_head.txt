104: static NTSTATUS ApexdrvOpenProcess(ULONG processId, HANDLE* processHandle) {
105:     CLIENT_ID clientId = { 0 };
106:     OBJECT_ATTRIBUTES attributes;
107: 
108:     clientId.UniqueProcess = ULongToHandle(processId);
109:     clientId.UniqueThread  = NULL;
110: 
111:     InitializeObjectAttributes(&attributes, NULL, OBJ_KERNEL_HANDLE, NULL, NULL);
112: 
113:     return ZwOpenProcess(processHandle,
114:                          PROCESS_CREATE_THREAD | PROCESS_VM_OPERATION |
115:                          PROCESS_VM_WRITE | PROCESS_VM_READ |
116:                          PROCESS_QUERY_INFORMATION,
117:                          &attributes, &clientId);
118: }

120: // ----------------------------------------------------------------------------
121: // IOCTL_APEXDRV_INJECT
122: // ----------------------------------------------------------------------------
123: static NTSTATUS ApexdrvInject(PAPEXDRV_INJECT_REQUEST request) {
124:     HANDLE   processHandle = NULL;
125:     HANDLE   threadHandle  = NULL;
126:     PVOID    remoteBuffer  = NULL;
127:     SIZE_T   regionSize    = 0;
128:     SIZE_T   pathBytes     = 0;
129:     NTSTATUS status;
130: 
131:     if (request->ProcessId == 0)          return STATUS_INVALID_PARAMETER;
132:     if (request->LoadLibraryAddress == 0)  return STATUS_INVALID_PARAMETER;
133: 
134:     // Ensure the wide path is NUL terminated inside the supplied buffer.
135:     request->DllPath[APEXDRV_MAX_PATH_CHARS - 1] = L'\0';
136:     pathBytes = (wcslen(request->DllPath) + 1) * sizeof(WCHAR);
137:     if (pathBytes <= sizeof(WCHAR)) return STATUS_INVALID_PARAMETER;
138: 
139:     status = ApexdrvOpenProcess(request->ProcessId, &processHandle);
140:     if (!NT_SUCCESS(status)) return status;
