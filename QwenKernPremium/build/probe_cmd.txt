699: // PHASE 1.1 in launcher form: measure, never inject.  Falls back to our own PID
700: // so it is runnable with the game closed (results are only comparable within
701: // the same process kind, hence the explicit self/target tag on the report).
702: int RunProbeCommand(DWORD pid) {
703:     const DWORD targetPid = (pid != 0) ? pid : GetCurrentProcessId();
704:     const bool  self      = (targetPid == GetCurrentProcessId());
705:     const char* tag       = self ? "self" : "target";
706: 
707:     printf("[*] AG measurement harness (%s PID %lu)%s\n", tag, targetPid,
708:            self ? " - game not running, measuring the launcher itself" : "");
709: 
710:     const ProbeEnvironment env = self ? AcgProbe::DescribeSelf()
711:                                       : AcgProbe::DescribeProcess(targetPid);
712:     printf("%s", AcgProbe::FormatEnvironment(env).c_str());
713: 
714:     // PHASE 1.3: reference facts for the same target, taken BEFORE any payload
715:     // is introduced, so "d3d11 is loaded" cannot be our own doing.
716:     if (!self) PrintReferenceIntegrity(CollectReferenceIntegrity(targetPid));
717: 
718:     const std::vector<ProbeResult> results =
719:         self ? AcgProbe::RunInternal() : AcgProbe::RunExternal(targetPid);
720:     printf("%s", AcgProbe::FormatReport("ACG measurement", env, results).c_str());
721: 
722:     int passed = 0;
723:     for (size_t i = 0; i < results.size(); ++i) {
724:         if (results[i].passed) ++passed;
725:     }
726: 
727:     // Advisory only - the ladder never reads this string.
728:     printf("[*] recommendation: %s\n", AcgProbe::Recommendation(results).c_str());
729:     printf("[*] measured %d/%d probe(s) passing\n", passed,
730:            static_cast<int>(results.size()));
731: 
732:     const std::string reportPath = AcgProbe::DefaultReportPath(tag);
733:     std::string error;
734:     if (AcgProbe::WriteReportFile(reportPath, "ACG measurement", env, results, error)) {
735:         printf("[*] report written: %s\n", reportPath.c_str());
736:     } else {
737:         printf("[!] report NOT written: %s\n", error.c_str());
738:     }
739: 
740:     return (passed == static_cast<int>(results.size())) ? 0 : 4;
741: }
