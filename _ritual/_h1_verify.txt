 113         W ('  prv ' + $id + ' no verdict (rc ignored by design)')

 119     # PatchGuard hashed this dword at boot. Restoring a constant (6) leaves the byte different ->
 120     # 0x109 CRITICAL_STRUCTURE_CORRUPTION ~20 min later (param4=0 generic data region, param3=this VA).
 121     $null = $script:lastOut -match 'value:\s*(\d+)\s*,\s*new value to be written'
 122     if($Matches){ $preDse = [int]$Matches[1] }

 124

 127     W '## STEP load: sc start apexdrv inside the open window'

 132

 140     $back = -1

 142     $restored = ($restored -and ($back -eq $preDse))

 144     W ('  restore verified=' + $restored + ' readback=' + $back + ' expect=' + $preDse)
 145 
 146     W ''
 147     W '=== VERDICT ==='
 148     if($loaded -and $restored){ W ('RESULT: PASS via prv ' + $winner + ' - driver RUNNING, DSE restored to boot value ' + $preDse + ' (readback verified), left RUNNING for h2'); W 'NEXT: run _ritual\h2_gate.ps1 (G1 gate run, 30 min) IN THIS BOOT - crash horizon ~20 min if the value is wrong' }

 150     else { W ('RESULT: FAIL via prv ' + $winner + ' - DSE written, load rc=' + $rcLoad) }
 151 } catch { W ('UNCAUGHT ' + $_.Exception.Message) }

 153     if($dseOpened -and $winner -ge 0 -and $preDse -ge 0){

 159         $null = RunExe (Join-Path $pay 'kdu.exe') @('-prv', "$winner", '-dse', '6') 20000
 160         W ('  finally restore verified=' + ($script:lastOut -match 'verification succeeded'))
 161     }
